+// Preserve each phase as an independent snapshot even when the caller reuses
+// its result variable, slice storage or SafeError pointer in later phases.
+func pr87SnapshotReconcile(result coreprovider.ReconcileResult) *coreprovider.ReconcileResult {
+	value := result
+	value.Observations = append([]coreprovider.ResourceObservation(nil), result.Observations...)
+	if result.Error != nil {
+		errCopy := *result.Error
+		value.Error = &errCopy
+	}
+	return &value
+}
+func pr87SnapshotCleanup(result coreprovider.OperationResult) *coreprovider.OperationResult {
+	value := result
+	value.ProviderResources = append([]coreprovider.ResourceResult(nil), result.ProviderResources...)
+	if result.Error != nil {
+		errCopy := *result.Error
+		value.Error = &errCopy
+	}
+	return &value
+}

+		initial := pr87SnapshotReconcile(result)

+		if !initial.Observations[0].Exists || initial.Observations[0].ObservedState != "ACTIVE" || final.Observations[0].Exists || initial == final {

+	})
