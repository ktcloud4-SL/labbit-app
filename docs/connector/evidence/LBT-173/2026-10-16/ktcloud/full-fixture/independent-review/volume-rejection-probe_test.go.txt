package openstackprovider

import (
 "context"
 "fmt"
 "net/http"
 "strings"
 "sync/atomic"
 "testing"
 "time"
 coreprovider "github.com/ktcloud4-SL/labbit-app/internal/connector/provider"
)

func TestReviewKTRejectedVolumeDeleteFreshState(t *testing.T) {
 cases := []struct{name string; code int; reality string; want coreprovider.Outcome; wantGets int32}{
  {"400absent",400,"absent",coreprovider.OutcomeSucceeded,0},
  {"409absent",409,"absent",coreprovider.OutcomeSucceeded,0},
  {"400deleting",400,"deleting",coreprovider.OutcomeSucceeded,0},
  {"409deleting",409,"deleting",coreprovider.OutcomeSucceeded,0},
  {"400stable",400,"stable",coreprovider.OutcomeFailed,2},
  {"409stable",409,"stable",coreprovider.OutcomeFailed,2},
  {"400lookupFailure",400,"lookupFailure",coreprovider.OutcomeUnknown,2},
  {"409malformed",409,"malformed",coreprovider.OutcomeUnknown,2},
  {"400wrongID",400,"wrongID",coreprovider.OutcomeUnknown,2},
  {"409attached",409,"attached",coreprovider.OutcomeUnknown,2},
  {"403noRetry",403,"absent",coreprovider.OutcomeFailed,1},
  {"500unknown",500,"absent",coreprovider.OutcomeUnknown,1},
  {"429unknown",429,"absent",coreprovider.OutcomeUnknown,1},
 }
 for _, tc := range cases { t.Run(tc.name,func(t *testing.T){
  var gets,deletes atomic.Int32
  a := ktTopologyAdapter(t,func(w http.ResponseWriter,r *http.Request){
   if !strings.HasSuffix(r.URL.Path,"/volumes/owned-root"){t.Error("request escaped tracked exact ID");w.WriteHeader(500);return}
   if r.Method==http.MethodDelete {deletes.Add(1);w.WriteHeader(tc.code);fmt.Fprint(w,`{"badRequest":{"code":400,"message":"REVIEW_PRIVATE_RAW_BODY"}}`);return}
   if r.Method!=http.MethodGet {t.Error("unexpected mutation");w.WriteHeader(500);return}
   n:=gets.Add(1)
   status,id,attachments:="available","owned-root",[]any{}
   if n>1 {
    switch tc.reality {
    case "absent":w.WriteHeader(404);return
    case "deleting":if n>2 {w.WriteHeader(404);return};status="deleting"
    case "stable":
    case "lookupFailure":w.WriteHeader(503);return
    case "malformed":fmt.Fprint(w,`{"volumeFault":{"code":500,"message":"REVIEW_PRIVATE_RAW_BODY"}}`);return
    case "wrongID":id="unrelated-root"
    case "attached":status="deleting";attachments=[]any{map[string]any{"server_id":"unrelated-server"}}
    }
   }
   writeJSON(t,w,200,map[string]any{"volume":map[string]any{"id":id,"status":status,"attachments":attachments}})
  })
  ctx,cancel:=context.WithTimeout(context.Background(),50*time.Millisecond);defer cancel()
  ref:=coreprovider.ResourceRef{ResourceType:coreprovider.ResourceTypeVolume,ProviderID:"owned-root",Generation:2,LogicalName:"workspace:root-volume"}
  result,err:=a.Cleanup(ctx,coreprovider.CleanupRequest{Correlation:coreprovider.Correlation{OperationID:"review-cleanup",LabInstanceID:"review-lab",Generation:2},ProviderResources:[]coreprovider.ResourceRef{ref}})
  t.Logf("outcome=%s GET=%d DELETE=%d",result.Outcome,gets.Load(),deletes.Load())
  if err!=nil || result.Outcome!=tc.want || deletes.Load()!=1 || (tc.wantGets!=0 && gets.Load()!=tc.wantGets) || (tc.wantGets==0 && gets.Load()<2){t.Fatalf("fresh exact state classification/retry boundary: outcome=%s want=%s GET=%d wantGets=%d DELETE=%d err=%v",result.Outcome,tc.want,gets.Load(),tc.wantGets,deletes.Load(),err)}
  if result.Error!=nil && strings.Contains(result.Error.Message,"REVIEW_PRIVATE_RAW_BODY"){t.Fatal("raw API body exposed")}
 }) }
}
