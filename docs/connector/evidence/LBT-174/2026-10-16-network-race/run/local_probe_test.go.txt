// External validation overlay only. Never copied into production or source archive.
func TestLocalLinuxRaceAndInternetDiagnostics(t *testing.T) {
	if os.Getenv("LABBIT_LOCAL_NETWORK_RACE") != "1" {
		t.Skip("explicit local validation opt-in required")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 55*time.Minute)
	defer cancel()
	a, err := New(ctx, ConfigFromEnvironment())
	if err != nil {
		t.Fatal("provider init failed")
	}
	images, err := a.ListImages(ctx)
	if err != nil {
		t.Fatal("image discovery failed")
	}
	flavors, err := a.ListFlavors(ctx)
	if err != nil {
		t.Fatal("flavor discovery failed")
	}
	networks, err := a.ListNetworks(ctx)
	if err != nil {
		t.Fatal("network discovery failed")
	}
	image := exactImage(t, images, "ubuntu")
	flavor := exactFlavor(t, flavors, "m1.small")
	management := exactNetwork(t, networks, "sharednet1")
	key := a.provision.SSHPrivateKeyFile
	keypair := integrationKeyPairName(t, ctx, a, key)
	sg := integrationManagementSecurityGroup(t, ctx, a, "172.16.8.1/32")
	external := integrationExternalNetworkID(t, ctx, a)
	a.provision = normalizedProvisionConfig(ProvisionConfig{ProviderConnectionID: "local-integration", ProjectID: integrationProjectID(t, a), ManagementNetworkID: management.ID, ManagementSecurityGroupID: sg, ExternalNetworkID: external, KeyPairName: keypair, SSHAllowedCIDR: "172.16.8.1/32", LabSubnetCIDR: "198.24.0.0/24", SSHUsername: "ubuntu", SSHPrivateKeyFile: key, SSHKnownHostsFile: t.TempDir() + "/known_hosts", PollInterval: 2 * time.Second})
	content := "#!/bin/sh\nset -eu\necho labbit-internet-policy-ready\n"
	digest := sha256.Sum256([]byte(content))
	snapshot := coreprovider.CreationSnapshot{ProviderConnectionID: "local-integration", WorkspaceVMKey: "workspace", InternetOutbound: true, StartupScript: &coreprovider.StartupScript{Content: content, SHA256: hex.EncodeToString(digest[:])}, VMs: []coreprovider.VMSpec{{VMKey: "workspace", Role: "WORKSPACE", ImageID: image.ID, FlavorID: flavor.ID, FlavorSpec: coreprovider.FlavorSpec{VCPUs: flavor.VCPUs, RAMMiB: flavor.RAMMiB, DiskGiB: flavor.DiskGiB}}}}
	labID := "lbt174-race-" + integrationRunID(t)
	result, err := a.Provision(ctx, coreprovider.ProvisionRequest{Correlation: coreprovider.Correlation{OperationID: labID + "-provision", LabInstanceID: labID, Generation: 1}, CreationSnapshot: snapshot})
	t.Cleanup(func() {
		cleanupM2Resources(t, a, result.ProviderResources)
		t.Logf("Linux race fixture exact tracked absence checked: resources=%d", len(result.ProviderResources))
	})
	if err != nil || result.Outcome != coreprovider.OutcomeSucceeded {
		t.Fatal("disposable fixture provision failed")
	}
	serverID := providerResourceID(t, result.ProviderResources, coreprovider.ResourceTypeServer, "workspace")
	managementID := providerResourceID(t, result.ProviderResources, coreprovider.ResourceTypePort, "workspace:management")
	p, err := ports.Get(ctx, a.network, managementID).Extract()
	if err != nil {
		t.Fatal("management lookup failed")
	}
	ip, err := managementAddress(*p)
	if err != nil {
		t.Fatal("management address unavailable")
	}
	host, err := sshHostKeyIdentity(a.provision.ProviderConnectionID, serverID)
	if err != nil {
		t.Fatal("host identity unavailable")
	}
	address := net.JoinHostPort(ip, "22")
	labPortID := providerResourceID(t, result.ProviderResources, coreprovider.ResourceTypePort, "workspace:lab")
	labPort, err := ports.Get(ctx, a.network, labPortID).Extract()
	if err != nil {
		t.Fatal("lab interface lookup failed")
	}
	if _, err := net.ParseMAC(labPort.MACAddress); err != nil {
		t.Fatal("lab MAC invalid")
	}
	if _, err := net.ParseMAC(p.MACAddress); err != nil {
		t.Fatal("management MAC invalid")
	}
	// No target connection or route probe precedes this exact first single TCP attempt.
	for attempt := 1; attempt <= 12; attempt++ {
		begin := time.Now()
		err := a.runSSHCommand(ctx, address, host, localReachabilityCommand(labPort.MACAddress, p.MACAddress))
		t.Logf("Internet first-series result attempt=%d success=%t elapsed_ms=%d", attempt, err == nil, time.Since(begin).Milliseconds())
	}
	client := localPinnedSSH(t, ctx, a, address, host)
	defer client.Close()
	out := os.Getenv("LABBIT_LOCAL_VALIDATION_DIR")
	phase := func(name, command string, limit time.Duration) error {
		t.Helper()
		session, err := client.NewSession()
		if err != nil {
			return errors.New("SSH phase channel failed")
		}
		defer session.Close()
		log, err := os.Create(filepath.Join(out, name+".log"))
		if err != nil {
			return err
		}
		defer log.Close()
		session.Stdout = log
		session.Stderr = log
		done := make(chan error, 1)
		go func() { done <- session.Run(command) }()
		timer := time.NewTimer(limit)
		defer timer.Stop()
		begin := time.Now()
		select {
		case err := <-done:
			status := 0
			var exit *ssh.ExitError
			if errors.As(err, &exit) {
				status = exit.ExitStatus()
			} else if err != nil {
				status = -1
			}
			t.Logf("Linux validation phase=%s exit=%d elapsed_ms=%d", name, status, time.Since(begin).Milliseconds())
			return err
		case <-timer.C:
			session.Close()
			return errors.New("Linux validation phase timeout")
		case <-ctx.Done():
			session.Close()
			return ctx.Err()
		}
	}
	if err := phase("route-counterfactual", localRouteABCommand(labPort.MACAddress, p.MACAddress), 45*time.Second); err != nil {
		t.Error("owned target-route counterfactual failed")
	}
	if err := phase("linux-preflight", "uname -sm; getconf GNU_LIBC_VERSION; command -v gcc || true; free -m; df -h /; sudo -n true", time.Minute); err != nil {
		t.Fatal("Linux preflight failed")
	}
	if err := phase("linux-apt", localAptCommand(labPort.MACAddress, p.MACAddress), 11*time.Minute); err != nil {
		t.Fatal("Linux C compiler preparation failed")
	}
	if err := phase("linux-directory", "mkdir -p /home/ubuntu/lbt174-validation", time.Minute); err != nil {
		t.Fatal("Linux staging failed")
	}
	fileclient, err := sftp.NewClient(client)
	if err != nil {
		t.Fatal("Linux SFTP unavailable")
	}
	defer fileclient.Close()
	for _, name := range []string{"go1.27.1.linux-amd64.tar.gz", "linux-source.tar.gz", "linux_ci.sh"} {
		input, err := os.Open(filepath.Join(out, name))
		if err != nil {
			t.Fatal("local staging input unavailable")
		}
		output, err := fileclient.Create("/home/ubuntu/lbt174-validation/" + name)
		if err != nil {
			input.Close()
			t.Fatal("Linux staging write failed")
		}
		_, copyErr := io.Copy(output, input)
		input.Close()
		closeErr := output.Close()
		if copyErr != nil || closeErr != nil {
			t.Fatal("Linux staging copy failed")
		}
	}
	install := fmt.Sprintf("cd /home/ubuntu/lbt174-validation && echo '%s  go1.27.1.linux-amd64.tar.gz' | sha256sum -c - && echo '%s  linux-source.tar.gz' | sha256sum -c - && tar xzf go1.27.1.linux-amd64.tar.gz && tar xzf linux-source.tar.gz", os.Getenv("LABBIT_LOCAL_GO_SHA256"), os.Getenv("LABBIT_LOCAL_SOURCE_SHA256"))
	if err := phase("linux-unpack", install, 3*time.Minute); err != nil {
		t.Fatal("Linux archive verification failed")
	}
	listener, err := client.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal("DB loopback forward unavailable")
	}
	defer listener.Close()
	go func() {
		for {
			remote, err := listener.Accept()
			if err != nil {
				return
			}
			go func() {
				defer remote.Close()
				local, err := net.DialTimeout("tcp", "127.0.0.1:15432", 5*time.Second)
				if err != nil {
					return
				}
				defer local.Close()
				done := make(chan struct{})
				go func() { io.Copy(local, remote); local.Close(); close(done) }()
				io.Copy(remote, local)
				remote.Close()
				<-done
			}()
		}
	}()
	_, port, err := net.SplitHostPort(listener.Addr().String())
	if err != nil {
		t.Fatal("DB forward port invalid")
	}
	command := "LABBIT_TEST_DATABASE_DSN='postgres://postgres@127.0.0.1:" + port + "/postgres?sslmode=disable' bash /home/ubuntu/lbt174-validation/linux_ci.sh"
	if err := phase("linux-ci", command, 35*time.Minute); err != nil {
		t.Fatal("Linux exact tagged-race CI phase failed; see bounded local phase output")
	}
	b, err := os.ReadFile(filepath.Join(out, "linux-ci.log"))
	if err != nil || !strings.Contains(string(b), "LBT174_LINUX_RACE_PASS") {
		t.Fatal("Linux race completion marker absent")
	}
	t.Log("Linux native CGO/race CI command completed; JSONL and exact required tests checked separately")
}

func localPython(script string) string {
	return "python3 -c \"import base64;exec(base64.b64decode('" + base64.StdEncoding.EncodeToString([]byte(script)) + "'))\""
}
func localNetworkPrelude(labMAC, managementMAC string) string {
	return fmt.Sprintf(`import json,socket,subprocess,time
from pathlib import Path
macs={'lab':%q,'management':%q}
def role(dev):
 try: value=(Path('/sys/class/net')/dev/'address').read_text().strip()
 except OSError: return 'unknown'
 return next((k for k,v in macs.items() if v.lower()==value.lower()),'unknown')
def route():
 rows=json.loads(subprocess.check_output(['ip','-j','route','get','1.1.1.1']))
 return [{'role':role(x.get('dev','')), 'hasGateway':'gateway' in x} for x in rows]
def probe():
 before=route();defaults=json.loads(subprocess.check_output(['ip','-j','route','show','default']))
 s=socket.socket(socket.AF_INET,socket.SOCK_STREAM);s.settimeout(5);start=time.monotonic();code=s.connect_ex(('1.1.1.1',443));s.close()
 return {'routeBefore':before,'defaultRoutes':[{'role':role(x.get('dev','')),'metric':x.get('metric',0),'hasGateway':'gateway' in x} for x in defaults],'connectCode':code,'elapsed_ms':round((time.monotonic()-start)*1000)}
`, labMAC, managementMAC)
}
func localReachabilityCommand(labMAC, managementMAC string) string {
	return localPython(localNetworkPrelude(labMAC, managementMAC) + "result=probe();print(json.dumps(result),flush=True);assert result['connectCode']==0, 'unexpected Internet reachability'\n")
}
func localRouteABCommand(labMAC, managementMAC string) string {
	return localPython(localNetworkPrelude(labMAC, managementMAC) + `defaults=json.loads(subprocess.check_output(['ip','-j','route','show','default']))
original=route()
assert not json.loads(subprocess.check_output(['ip','-j','route','show','exact','1.1.1.1/32'])), 'target host route already exists'
print(json.dumps({'case':'baseline',**probe()}),flush=True)
for wanted in ['management','lab']:
 selected=next(x for x in defaults if role(x.get('dev',''))==wanted and 'gateway' in x)
 args=['1.1.1.1/32','via',selected['gateway'],'dev',selected['dev'],'metric','50']
 subprocess.run(['sudo','ip','route','add']+args,check=True,stdout=subprocess.DEVNULL)
 try:
  result=probe();print(json.dumps({'case':'forced-'+wanted,**result}),flush=True)
  assert result['routeBefore'][0]['role']==wanted
  assert (result['connectCode']==0)==(wanted=='lab')
 finally: subprocess.run(['sudo','ip','route','del']+args,check=True,stdout=subprocess.DEVNULL)
assert not json.loads(subprocess.check_output(['ip','-j','route','show','exact','1.1.1.1/32']))
assert route()==original
print(json.dumps({'case':'restored','route':route(),'ownedRouteAbsent':True}),flush=True)
`)
}
func localAptCommand(labMAC, managementMAC string) string {
	return localPython(localNetworkPrelude(labMAC, managementMAC) + `defaults=json.loads(subprocess.check_output(['ip','-j','route','show','default']))
selected=next(x for x in defaults if role(x.get('dev',''))=='lab' and 'gateway' in x)
args=['default','via',selected['gateway'],'dev',selected['dev'],'metric','40']
subprocess.run(['sudo','ip','route','add']+args,check=True,stdout=subprocess.DEVNULL)
try:
 subprocess.run(['sudo','timeout','300s','apt-get','update','-qq'],check=True)
 subprocess.run(['sudo','env','DEBIAN_FRONTEND=noninteractive','timeout','300s','apt-get','install','-y','--no-install-recommends','gcc','libc6-dev'],check=True)
finally: subprocess.run(['sudo','ip','route','del']+args,check=True,stdout=subprocess.DEVNULL)
`)
}

func localPinnedSSH(t *testing.T, ctx context.Context, a *Adapter, address, host string) *ssh.Client {
	t.Helper()
	key, err := os.ReadFile(a.provision.SSHPrivateKeyFile)
	if err != nil {
		t.Fatal("private key input unavailable")
	}
	signer, err := ssh.ParsePrivateKey(key)
	if err != nil {
		t.Fatal("private key input invalid")
	}
	callback, err := trustOnFirstUseCallback(a.provision.SSHKnownHostsFile, host)
	if err != nil {
		t.Fatal("host pin unavailable")
	}
	tcp, err := (&net.Dialer{Timeout: 5 * time.Second}).DialContext(ctx, "tcp", address)
	if err != nil {
		t.Fatal("SSH dial failed")
	}
	t.Cleanup(func() { tcp.Close() })
	stop := context.AfterFunc(ctx, func() { tcp.Close() })
	t.Cleanup(func() { stop() })
	conn, channels, requests, err := ssh.NewClientConn(tcp, host, &ssh.ClientConfig{User: a.provision.SSHUsername, Auth: []ssh.AuthMethod{ssh.PublicKeys(signer)}, HostKeyCallback: callback})
	if err != nil {
		tcp.Close()
		t.Fatal("SSH authentication or host pin failed")
	}
	return ssh.NewClient(conn, channels, requests)
}
