func TestManagementDHCPPolicyRejectsUnsupportedDeploymentBeforeMutation(t *testing.T) {
	for _, test := range []struct {
		name string
		path string
		body map[string]any
		ssh  string
		lab  string
	}{
		{name: "extension absent", path: "/extensions/extra_dhcp_opt", body: map[string]any{"extension": map[string]any{"alias": "other"}}},
		{name: "multiple subnets", path: "/networks/management-network", body: map[string]any{"network": map[string]any{"id": "management-network", "status": "ACTIVE", "subnets": []string{"management-subnet", "second"}}}},
		{name: "network identity mismatch", path: "/networks/management-network", body: map[string]any{"network": map[string]any{"id": "other", "status": "ACTIVE", "subnets": []string{"management-subnet"}}}},
		{name: "network unavailable", path: "/networks/management-network", body: map[string]any{"network": map[string]any{"id": "management-network", "status": "DOWN", "subnets": []string{"management-subnet"}}}},
		{name: "routed SSH", ssh: "172.16.9.1/32"},
		{name: "SSH prefix only partly contained", ssh: "172.16.0.0/16"},
		{name: "Lab overlaps management", lab: "172.16.8.0/25"},
		{name: "DHCP disabled", path: "/subnets/management-subnet", body: invalidManagementSubnet("enable_dhcp", false)},
		{name: "IPv6", path: "/subnets/management-subnet", body: invalidManagementSubnet("ip_version", 6)},
		{name: "invalid CIDR", path: "/subnets/management-subnet", body: invalidManagementSubnet("cidr", "invalid")},
		{name: "subnet identity mismatch", path: "/subnets/management-subnet", body: invalidManagementSubnet("id", "other")},
		{name: "subnet parent mismatch", path: "/subnets/management-subnet", body: invalidManagementSubnet("network_id", "other")},
		{name: "default host route", path: "/subnets/management-subnet", body: invalidManagementSubnet("host_routes", []any{map[string]any{"destination": "0.0.0.0/0", "nexthop": "172.16.8.2"}})},
		{name: "specific host route dependency", path: "/subnets/management-subnet", body: invalidManagementSubnet("host_routes", []any{map[string]any{"destination": "192.0.2.0/24", "nexthop": "172.16.8.2"}})},
	} {
		t.Run(test.name, func(t *testing.T) {
			fake := &m2OpenStackFake{t: t}
			mutations := 0
			a := newTestAdapter(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				if r.Method != http.MethodGet {
					mutations++
				}
				if test.path != "" && strings.HasSuffix(r.URL.Path, test.path) {
					writeJSON(t, w, http.StatusOK, test.body)
					return
				}
				fake.ServeHTTP(w, r)
			}))
			a.provision = testProvisionConfig(t)
			if test.ssh != "" {
				a.provision.SSHAllowedCIDR = test.ssh
			}
			if test.lab != "" {
				a.provision.LabSubnetCIDR = test.lab
			}
			result, err := a.Provision(context.Background(), coreprovider.ProvisionRequest{Correlation: coreprovider.Correlation{OperationID: "route-preflight", LabInstanceID: "lab", Generation: 1}, CreationSnapshot: validSnapshot()})
			if err != nil || result.Outcome != coreprovider.OutcomeFailed || result.Error == nil || result.Error.Code != errorOpenStack || mutations != 0 || len(result.ProviderResources) != 0 {
				t.Fatalf("preflight mutations=%d result=%+v err=%v", mutations, result, err)
			}
		})
	}
}

func invalidManagementSubnet(field string, value any) map[string]any {
	s := map[string]any{"id": "management-subnet", "network_id": "management-network", "ip_version": 4, "enable_dhcp": true, "cidr": "172.16.8.0/24", "gateway_ip": "172.16.8.2", "host_routes": []any{}}
	s[field] = value
	return map[string]any{"subnet": s}
}

func TestManagementDHCPPolicyAppliesToOnlyOwnedManagementPortAndPreservesUserData(t *testing.T) {
	for _, online := range []bool{true, false} {
		t.Run(map[bool]string{true: "on", false: "off"}[online], func(t *testing.T) {
			fake := &m2OpenStackFake{t: t}
			a := newTestAdapter(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				if r.Method != http.MethodGet && strings.Contains(r.URL.Path, "management-") {
					t.Fatalf("unexpected shared mutation: %s %s", r.Method, r.URL.Path)
				}
				if r.Method == http.MethodPost && strings.HasSuffix(r.URL.Path, "/ports") {
					var body struct {
						Port map[string]json.RawMessage `json:"port"`
					}
					// Inspect at the boundary while retaining the original request for the fake.
					b, err := io.ReadAll(r.Body)
					if err != nil {
						t.Fatal(err)
					}
					r.Body = io.NopCloser(strings.NewReader(string(b)))
					if err := json.Unmarshal(b, &body); err != nil {
						t.Fatal(err)
					}
					var network string
					_ = json.Unmarshal(body.Port["network_id"], &network)
					if network == "management-network" {
						var opts []managementDHCPOption
						if err := json.Unmarshal(body.Port["extra_dhcp_opts"], &opts); err != nil || !managementRoutesSuppressed(opts) {
							t.Fatalf("management opts do not suppress routes: %s", body.Port["extra_dhcp_opts"])
						}
						if !strings.Contains(string(body.Port["fixed_ips"]), "management-subnet") {
							t.Fatal("management subnet not pinned")
						}
					} else if _, ok := body.Port["extra_dhcp_opts"]; ok {
						t.Fatal("Lab DHCP changed")
					}
				}
				fake.ServeHTTP(w, r)
			}))
			a.provision = testProvisionConfig(t)
			a.sshProbe = func(context.Context, string) error { return nil }
			a.startupProbe = func(context.Context, string, string) error { return nil }
			snapshot := validSnapshot()
			snapshot.InternetOutbound = online
			result, err := a.Provision(context.Background(), coreprovider.ProvisionRequest{Correlation: coreprovider.Correlation{OperationID: "route-policy", LabInstanceID: "lab", Generation: 1}, CreationSnapshot: snapshot})
			if err != nil || result.Outcome != coreprovider.OutcomeSucceeded || fake.userData != snapshot.StartupScript.Content {
				t.Fatalf("policy result=%+v err=%v", result, err)
			}
		})
	}
}

func TestManagementDHCPPolicyKeepsCreatedPortWhenProviderDropsOverride(t *testing.T) {
	for _, stage := range []string{"create", "read"} {
		t.Run(stage, func(t *testing.T) {
			fake := &m2OpenStackFake{t: t, omitManagementDHCPOnCreate: stage == "create", omitManagementDHCPOnGet: stage == "read"}
			a := newTestAdapter(t, fake)
			a.provision = testProvisionConfig(t)
			result, err := a.Provision(context.Background(), coreprovider.ProvisionRequest{Correlation: coreprovider.Correlation{OperationID: "route-drop", LabInstanceID: "lab", Generation: 1}, CreationSnapshot: validSnapshot()})
			found := false
			for _, r := range result.ProviderResources {
				if r.ProviderID == "port-management" {
					found = true
				}
			}
			if err != nil || result.Outcome == coreprovider.OutcomeSucceeded || !found || fake.serverCreates != 0 {
				t.Fatalf("lost partial port: result=%+v creates=%d err=%v", result, fake.serverCreates, err)
			}
		})
	}
}

func TestManagementDHCPPolicyRejectsMalformedOptions(t *testing.T) {
	empty, nonempty := "", "172.16.8.2"
	valid := func() []managementDHCPOption {
		return []managementDHCPOption{{Name: "router", Value: &empty, Version: 4}, {Name: "classless-static-route", Value: &empty, Version: 4}}
	}
	for _, kind := range []string{"missing", "null", "duplicate", "IPv6", "nonempty", "unknown", "numeric name", "unsupported 249"} {
		t.Run(kind, func(t *testing.T) {
			opts := valid()
			switch kind {
			case "missing":
				opts = opts[:1]
			case "null":
				opts[0].Value = nil
			case "duplicate":
				opts[1].Name = "router"
			case "IPv6":
				opts[0].Version = 6
			case "nonempty":
				opts[0].Value = &nonempty
			case "numeric name":
				opts[0].Name = "3"
			case "unsupported 249":
				opts = append(opts, managementDHCPOption{Name: "249", Value: &empty, Version: 4})
			case "unknown":
				opts[0].Name = "unknown"
			}
			if managementRoutesSuppressed(opts) {
				t.Fatalf("accepted %s options", kind)
			}
		})
	}
	if !managementRoutesSuppressed(valid()) {
		t.Fatal("explicit empty overrides rejected")
	}
}

func TestManagementDHCPPolicyResetPreflightRetainsExistingGeneration(t *testing.T) {
	base := &m2OpenStackFake{t: t}
	fake := &m3ResetFake{base: base, deleted: make(map[string]bool)}
	reject := false
	checked := false
	a := newTestAdapter(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if reject && strings.HasSuffix(r.URL.Path, "/extensions/extra_dhcp_opt") {
			checked = true
			http.NotFound(w, r)
			return
		}
		fake.ServeHTTP(w, r)
	}))
	a.provision = testProvisionConfig(t)
	a.sshProbe = func(context.Context, string) error { return nil }
	a.startupProbe = func(context.Context, string, string) error { return nil }
	snapshot := validSnapshot()
	first, err := a.Provision(context.Background(), coreprovider.ProvisionRequest{Correlation: coreprovider.Correlation{OperationID: "first", LabInstanceID: "lab", Generation: 1}, CreationSnapshot: snapshot})
	if err != nil || first.Outcome != coreprovider.OutcomeSucceeded {
		t.Fatalf("initial provision=%+v err=%v", first, err)
	}
	refs := resourceRefs(first.ProviderResources, 1)
	reject = true
	reset, err := a.Reset(context.Background(), coreprovider.ResetRequest{Correlation: coreprovider.Correlation{OperationID: "reset", LabInstanceID: "lab", Generation: 2}, CreationSnapshot: snapshot, ProviderResources: refs})
	if err != nil || !checked || reset.Outcome != coreprovider.OutcomeFailed || len(fake.deleteOrder) != 0 || len(reset.ProviderResources) != len(refs) {
		t.Fatalf("reset=%+v checked=%t deletes=%v err=%v", reset, checked, fake.deleteOrder, err)
	}
	for i, r := range reset.ProviderResources {
		if r.ResourceRef != refs[i] || r.ObservedState != stateDeleteNotAttempted {
			t.Fatal("existing generation mutated during rejected preflight")
		}
	}
}

func TestManagementDHCPPolicyLookupFailureAndCancellationNeverMutate(t *testing.T) {
	for _, mode := range []string{"unauthorized", "unavailable", "canceled"} {
		t.Run(mode, func(t *testing.T) {
			mutations := 0
			ctx, cancel := context.WithTimeout(context.Background(), time.Second)
			defer cancel()
			a := newTestAdapter(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				if r.Method != http.MethodGet {
					mutations++
				}
				if mode == "canceled" {
					cancel()
					<-r.Context().Done()
					return
				}
				status := http.StatusForbidden
				if mode == "unavailable" {
					status = http.StatusInternalServerError
				}
				http.Error(w, "sensitive provider diagnostic must not escape", status)
			}))
			a.provision = testProvisionConfig(t)
			result, err := a.Provision(ctx, coreprovider.ProvisionRequest{Correlation: coreprovider.Correlation{OperationID: "failure", LabInstanceID: "lab", Generation: 1}, CreationSnapshot: validSnapshot()})
			if err != nil || result.Outcome == coreprovider.OutcomeSucceeded || mutations != 0 || len(result.ProviderResources) != 0 || result.Error == nil || strings.Contains(result.Error.Message, "sensitive") {
				t.Fatalf("lookup result=%+v mutations=%d err=%v", result, mutations, err)
			}
		})
	}
}
