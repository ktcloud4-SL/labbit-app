package openstackprovider

import (
	"encoding/base64"
	"encoding/json"
	"os"
	"strings"
	"testing"
)

// Calls the as-run Go generators without connecting to any VM or executing Python.
func TestReviewDumpGeneratedRemotePython(t *testing.T) {
	const lab = "02:00:00:00:00:01"
	const management = "02:00:00:00:00:02"
	commands := map[string]string{
		"reachability": localReachabilityCommand(lab, management),
		"route-ab": localRouteABCommand(lab, management, "192.0.2.1"),
		"apt": localAptCommand(lab, management),
		"intentional-timeout": localTimeoutCleanupCommand(lab, management),
		"renew-lab-first": localRenewCommand(lab, management, "lab-first"),
		"renew-management-first": localRenewCommand(lab, management, "management-first"),
	}
	decoded := map[string]string{}
	const prefix = "python3 -c \"import base64;exec(base64.b64decode('"
	const suffix = "'))\""
	for name, command := range commands {
		if !strings.HasPrefix(command, prefix) || !strings.HasSuffix(command, suffix) {
			t.Fatalf("%s command wrapper changed", name)
		}
		payload := strings.TrimSuffix(strings.TrimPrefix(command, prefix), suffix)
		script, err := base64.StdEncoding.DecodeString(payload)
		if err != nil {
			t.Fatalf("%s: %v", name, err)
		}
		decoded[name] = string(script)
	}
	data, err := json.MarshalIndent(decoded, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	path := os.Getenv("LBT174_REVIEW_PYTHON_DUMP")
	if path == "" {
		t.Fatal("LBT174_REVIEW_PYTHON_DUMP is required")
	}
	if err := os.WriteFile(path, append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("dumped %d generated remote Python scripts; no commands executed", len(decoded))
}
