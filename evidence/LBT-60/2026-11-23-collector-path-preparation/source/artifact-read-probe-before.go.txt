type event struct {
	API, Region, Result string
	ElapsedMS           int64
}
type recorder struct{ rows []event }

func (r *recorder) Run(c context.Context, e string, args []string) ([]byte, error) {
	at := time.Now()
	b, err := (drcontrol.CommandRunner{}).Run(c, e, args)
	region := ""
	for i := range args {
		if args[i] == "--region" && i+1 < len(args) {
			region = args[i+1]
		}
	}
	state := "RESPONSE_RECEIVED"
	if err != nil {
		state = "NOT_VERIFIED"
	}
	r.rows = append(r.rows, event{API: args[0] + "/" + args[1], Region: region, Result: state, ElapsedMS: time.Since(at).Milliseconds()})
	return b, err
}
