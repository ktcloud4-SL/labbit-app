import copy
import importlib.util
from pathlib import Path
import unittest
from unittest.mock import patch

spec = importlib.util.spec_from_file_location('policy', Path(__file__).with_name('pr196_collector_policy.py'))
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)

class PolicySafety(unittest.TestCase):
    def fixture(self):
        c = {'resourcesVpcConfig': {'vpcId': 'vpc-pinned', 'subnetIds': ['subnet-a', 'subnet-b'], 'clusterSecurityGroupId': 'sg-cluster'}}
        def eni(ip):
            return {'Description': 'Amazon EKS labbit-seoul-cluster', 'RequesterManaged': True, 'InterfaceType': 'interface', 'Status': 'in-use', 'VpcId': 'vpc-pinned', 'SubnetId': 'subnet-a', 'Groups': [{'GroupId': 'sg-cluster'}], 'Attachment': {'Status': 'attached', 'InstanceOwnerId': 'service-account'}, 'PrivateIpAddress': ip}
        return c, [eni('10.0.1.2'), eni('10.0.2.3')]

    def test_managed_node_same_security_group_not_selected(self):
        c, es = self.fixture()
        node = copy.deepcopy(es[0]); node['Description'] = 'Amazon EKS node'; node['PrivateIpAddress'] = '10.0.3.4'
        self.assertEqual(m.control_plane_ips(c, es + [node]), ['10.0.1.2', '10.0.2.3'])

    def test_matching_description_requires_all_ownership_checks(self):
        cases = [('RequesterManaged', False), ('InterfaceType', 'trunk'), ('Status', 'available'), ('VpcId', 'other'), ('SubnetId', 'other'), ('PrivateIpAddress', '8.8.8.8'), ('Groups', [{'GroupId': 'other'}]), ('Attachment', {'Status': 'attached', 'InstanceId': 'i-node'}), ('Attachment', {'Status': 'attached', 'InstanceOwnerId': '299311846289'})]
        for key, value in cases:
            with self.subTest(key=key, value=value):
                c, es = self.fixture(); es[0][key] = value
                with self.assertRaises(RuntimeError): m.control_plane_ips(c, es)

    def test_missing_duplicate_or_extra_sources_rejected(self):
        c, es = self.fixture()
        for values in [es[:1], [es[0], es[0]], es + [dict(es[0], PrivateIpAddress='10.0.3.4')]]:
            with self.assertRaises(RuntimeError): m.control_plane_ips(c, values)

    def test_policy_exact_sources_backend_and_admin_port_only(self):
        p = m.build_policy(['10.0.1.2', '10.0.2.3'])
        self.assertEqual(p['spec']['ingress'], [{'from': [{'ipBlock': {'cidr': '10.0.1.2/32'}}, {'ipBlock': {'cidr': '10.0.2.3/32'}}], 'ports': [{'protocol': 'TCP', 'port': 9090}]}])
        self.assertEqual(p['spec']['policyTypes'], ['Ingress'])
        self.assertEqual(len(p['spec']['podSelector']['matchLabels']), 2)
        for ips in [['0.0.0.0/0', '10.0.2.3'], ['10.0.1.2'], ['10.0.1.2', '10.0.1.2']]:
            with self.assertRaises((ValueError, RuntimeError)): m.build_policy(ips)

    def test_cleanup_wrong_uid_label_or_spec_never_deletes(self):
        original = m.build_policy(['10.0.1.2', '10.0.2.3'])
        original['metadata'].update(uid='own', resourceVersion='7')
        for kind in ['uid', 'label', 'spec']:
            with self.subTest(kind=kind):
                actual = copy.deepcopy(original)
                if kind == 'uid': actual['metadata']['uid'] = 'recreated'
                elif kind == 'label': actual['metadata']['labels']['labbit.io/trial'] = 'other'
                else: actual['spec']['ingress'] = []
                with patch.object(m.r, 'identity'), patch.object(m.r, 'load', return_value={'result': original}), patch.object(m.r, 'kube', return_value=actual) as k, patch.object(m.r, 'save') as s:
                    with self.assertRaises(RuntimeError): m.cleanup()
                    self.assertEqual(k.call_count, 1); s.assert_not_called()

if __name__ == '__main__': unittest.main(verbosity=2)
