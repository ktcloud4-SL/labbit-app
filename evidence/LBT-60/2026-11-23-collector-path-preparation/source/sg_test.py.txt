import copy
import importlib.util
from pathlib import Path
import unittest
from unittest.mock import patch
sp=importlib.util.spec_from_file_location('window',Path(__file__).with_name('pr196_collector_network_window2.py'))
m=importlib.util.module_from_spec(sp);sp.loader.exec_module(m)

class SGSafety(unittest.TestCase):
    def rule(self):
        return {'GroupId':m.GROUP,'GroupOwnerId':'299311846289','IsEgress':False,'IpProtocol':'tcp','FromPort':9090,'ToPort':9090,'CidrIpv4':'10.0.1.2/32','Description':m.DESCRIPTION,'SecurityGroupRuleId':'sgr-own'}

    def test_exact_source32_admin_only_permissions(self):
        p=m.permission(['10.0.1.2','10.0.2.3'])[0]
        self.assertEqual((p['IpProtocol'],p['FromPort'],p['ToPort']),('tcp',9090,9090))
        self.assertEqual([v['CidrIp'] for v in p['IpRanges']],['10.0.1.2/32','10.0.2.3/32'])
        for ips in [['0.0.0.0/0','10.0.2.3'],['8.8.8.8','10.0.2.3'],['10.0.1.2','10.0.1.2']]:
            with self.assertRaises((ValueError,RuntimeError)):m.permission(ips)

    def test_wrong_owner_port_egress_source_description_and_id_rejected(self):
        for k,v in [('GroupId','sg-other'),('GroupOwnerId','000000000000'),('IsEgress',True),('IpProtocol','-1'),('FromPort',8080),('ToPort',65535),('CidrIpv4','10.0.0.0/16'),('Description','other'),('SecurityGroupRuleId','')]:
            with self.subTest(key=k):
                r=self.rule();r[k]=v
                with self.assertRaises(RuntimeError):m.verify_rule(r,['10.0.1.2','10.0.2.3'])
        m.verify_rule(self.rule(),['10.0.1.2','10.0.2.3'])

    def test_cleanup_wrong_rule_set_never_revokes(self):
        a=self.rule();b=dict(a,SecurityGroupRuleId='sgr-two',CidrIpv4='10.0.2.3/32')
        def load(_,name):
            return {'result':{'SecurityGroupRules':[a,b]}} if name.endswith('receipt.json') else {'source_ips':['10.0.1.2','10.0.2.3']}
        wrong=dict(b,Description='other')
        with patch.object(m.r,'identity'),patch.object(m.r,'load',side_effect=load),patch.object(m.r,'aws',return_value={'SecurityGroupRules':[a,wrong]}) as aws,patch.object(m.r,'save') as save:
            with self.assertRaises(RuntimeError):m.sg_cleanup()
            self.assertEqual(aws.call_count,1);save.assert_not_called()

    def test_cleanup_only_saved_ids(self):
        a=self.rule();b=dict(a,SecurityGroupRuleId='sgr-two',CidrIpv4='10.0.2.3/32')
        def load(_,name):return {'result':{'SecurityGroupRules':[a,b]}} if name.endswith('receipt.json') else {'source_ips':['10.0.1.2','10.0.2.3']}
        with patch.object(m.r,'identity'),patch.object(m.r,'load',side_effect=load),patch.object(m.r,'aws',side_effect=[{'SecurityGroupRules':[a,b]},{'Return':True}]) as aws,patch.object(m.r,'save'):
            m.sg_cleanup()
            self.assertEqual(aws.call_args.args,('ec2','revoke-security-group-ingress','--group-id',m.GROUP,'--security-group-rule-ids','sgr-own','sgr-two'))

    def test_revoke_false_preserved_and_never_claimed_success(self):
        a=self.rule();b=dict(a,SecurityGroupRuleId='sgr-two',CidrIpv4='10.0.2.3/32')
        def load(_,name):return {'result':{'SecurityGroupRules':[a,b]}} if name.endswith('receipt.json') else {'source_ips':['10.0.1.2','10.0.2.3']}
        for response in [{'Return':False},{}]:
            with self.subTest(response=response),patch.object(m.r,'identity'),patch.object(m.r,'load',side_effect=load),patch.object(m.r,'aws',side_effect=[{'SecurityGroupRules':[a,b]},response]),patch.object(m.r,'save') as save,patch('builtins.print') as output:
                with self.assertRaisesRegex(RuntimeError,'REVOKE_NOT_CONFIRMED_HOLD'):m.sg_cleanup()
                self.assertEqual(save.call_args.args[1],'sg-delete.receipt.json');output.assert_not_called()

if __name__=='__main__':unittest.main(verbosity=2)
