 if p.stat().st_size>150000:
  raw=p.read_bytes();target=p.with_suffix(p.suffix+'.gz')
  target.write_bytes(gzip.compress(raw,mtime=0))
  compressions.append({'original':p.name,'compressed':target.name,'original_bytes':len(raw),'original_sha256':sha(raw),'compressed_sha256':sha(target.read_bytes())})

save('log-compression.json',compressions)

def countlog(name):
 p=OUT/(name+'.log'); gz=p.with_suffix('.log.gz')
 raw=gzip.decompress(gz.read_bytes()) if gz.exists() else p.read_bytes()
 counts={'pass':0,'fail':0,'skip':0};packages={};skips=[]
 for line in raw.decode('utf-8-sig').splitlines():
  try:r=json.loads(line)
  except ValueError:continue
  action=r.get('Action')
  if r.get('Test') and action in counts:
   counts[action]+=1
   if action=='skip':skips.append({'package':r['Package'],'test':r['Test']})
  elif not r.get('Test') and action in ('pass','fail','skip'):packages[r['Package']]=action
 return {'counts':counts,'packages':packages,'skips':skips,'log':gz.name if gz.exists() else p.name}
