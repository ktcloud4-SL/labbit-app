// Read-only smoke harness, not an operator/approval tool. No SQL/S3/mutation.
package main

import (
	"context"
	"encoding/json"
	"os"
	"time"

	"github.com/ktcloud4-SL/labbit-app/internal/drcontrol"
)

type readOnly struct{ calls int }

func (r *readOnly) Run(ctx context.Context, path string, args []string) ([]byte, error) {
	if len(args) < 2 || !((args[0] == "sts" && args[1] == "get-caller-identity") || (args[0] == "rds" && args[1] == "describe-db-instances")) {
		return nil, drcontrol.ErrDenied
	}
	r.calls++
	return (drcontrol.CommandRunner{}).Run(ctx, path, args)
}

func main() {
	out := map[string]any{"execution_gate": "HOLD", "recovery_authorized": false, "cloud_verified": false, "scope": "readonly AWS transport and RDS metadata only; synthetic policy trust fields; no SQL/approval/Resume"}
	code := 2
	defer func() { _ = json.NewEncoder(os.Stdout).Encode(out); os.Exit(code) }()
	if len(os.Args) != 2 {
		out["reason"] = "INPUT_NOT_VERIFIED"
		return
	}
	b, err := os.ReadFile(os.Args[1])
	var p drcontrol.Policy
	if err != nil || drcontrol.Strict(b, &p) != nil || p.Validate() != nil || p.EnableExecution {
		out["reason"] = "POLICY_NOT_VERIFIED"
		return
	}
	runner := &readOnly{}
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	client, err := drcontrol.NewAWSCLI(p, runner)
	if err != nil {
		out["reason"] = "ADAPTER_NOT_VERIFIED"
		return
	}
	started := time.Now().UTC()
	out["started_at_utc"] = started
	id, err := client.Identity(ctx)
	if err != nil || id.Account != p.AccountID || len(p.Actors) != 1 || id.ARN != p.Actors[0].ARN {
		out["reason"] = "IDENTITY_NOT_VERIFIED"
		return
	}
	out["identity_metadata"] = "MATCH"
	if err = client.VerifyDatabase(ctx); err != nil {
		out["reason"] = "RDS_METADATA_NOT_VERIFIED"
		return
	}
	out["rds_metadata"] = "MATCH"
	p.Database.ResourceID += "-intentionally-wrong"
	wrong, err := drcontrol.NewAWSCLI(p, runner)
	if err != nil || wrong.VerifyDatabase(ctx) != drcontrol.ErrDenied {
		out["reason"] = "MISMATCH_REJECTION_NOT_VERIFIED"
		return
	}
	out["wrong_resource_id"] = "REJECTED"
	out["external_calls"] = runner.calls
	out["finished_at_utc"] = time.Now().UTC()
	out["reason"] = "READONLY_METADATA_SMOKE_PASSED"
	code = 0
}
