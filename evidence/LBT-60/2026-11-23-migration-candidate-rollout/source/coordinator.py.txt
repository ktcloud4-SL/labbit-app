def now():
    return dt.datetime.now(dt.timezone.utc).isoformat()


def save(folder, name, data):
    folder.mkdir(parents=True, exist_ok=True)
    with (folder / name).open('x', encoding='utf-8', newline='\n') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
        f.write('\n')


def load(folder, name):
    return json.loads((folder / name).read_text(encoding='utf-8-sig'))


def command(args, timeout=30, text=False):
    env = {k: v for k, v in os.environ.items() if not k.upper().startswith(('AWS_', 'PG'))}
    env.update(AWS_PROFILE='labbit-admin3', AWS_PAGER='', AWS_CLI_AUTO_PROMPT='off', AWS_MAX_ATTEMPTS='1', AWS_IGNORE_CONFIGURED_ENDPOINT_URLS='true')
    p = subprocess.run(args, env=env, capture_output=True, timeout=timeout)
    if p.returncode:
        # Do not copy arbitrary cloud/DB raw errors into evidence or conversation.
        raise RuntimeError('COMMAND_FAILED_EXIT_' + str(p.returncode))
    return p.stdout.decode('utf-8-sig') if text else json.loads(p.stdout)


def aws(service, action, *args):

def check_db(stage, expected_version):
    x = load(EVIDENCE, 'db-' + stage + '.json')
    if x['status'] != 'OBSERVED_READONLY' or not x.get('local_port_closed') or not x.get('session_history_terminated') or not x.get('session_active_absent'):
        raise RuntimeError('DB_OR_SESSION_NOT_VERIFIED')
    age = (dt.datetime.now(dt.timezone.utc) - dt.datetime.fromisoformat(x['identity']['observed_at'])).total_seconds()
    if age < 0 or age > 300:
        raise RuntimeError('DB_OBSERVATION_STALE')
    if x.get('secret_reference_version') != SECRET_VERSION or not x.get('migration_startup_timeouts_verified'):
        raise RuntimeError('SECRET_VERSION_OR_TIMEOUTS_NOT_VERIFIED')
    rows = x['migrations']
    if [r['version'] for r in rows] != list(range(1, expected_version + 1)):
        raise RuntimeError('LEDGER_VERSION_MISMATCH')
    for row in rows:
        b = subprocess.check_output(['git', 'show', 'be4d0ba:db/migrations/' + row['name']], cwd=ROOT)
        if hashlib.sha256(b).hexdigest() != row['checksum']:
            raise RuntimeError('LEDGER_CHECKSUM_MISMATCH')
    inv = x['work_inventory'][0]
    if any(inv[k] != 0 for k in ('unfinished_operations', 'unfinished_items', 'unresolved_provider_records', 'active_terminal_sessions', 'other_open_transactions')):
        raise RuntimeError('WORK_NOT_IDLE')
    if expected_version == 15:
        if len(x['control_table_counts']) != 5 or any(t['rows'] for t in x['control_table_counts']):
            raise RuntimeError('CONTROL_JOURNAL_USED_NO_AUTOMATIC_LEGACY_ROLLBACK')
        if not any(c['conname'] == 'worker_control_resume_applied_time' and c['convalidated'] for c in x['resume_constraints']):
            raise RuntimeError('MIGRATION14_CONSTRAINT_NOT_VERIFIED')
    return x

def patch(resource, namespace, operations):
    path = LOCAL / ('patch-' + str(dt.datetime.now().timestamp()).replace('.', '-') + '.json')
    path.write_text(json.dumps(operations), encoding='utf-8')
    return kube('patch', resource, '-n', namespace, '--type=json', '--patch-file', str(path), '-o', 'json')

def observe(label):
    d=deployment()
    rs=kube('get','replicasets','-n','labbit','-o','json')['items']
    owned={r['metadata']['uid'] for r in rs if any(o['uid']==UID for o in r['metadata'].get('ownerReferences',[]))}
    pods=kube('get','pods','-n','labbit','-o','json')['items']
    result=[]
    for p in pods:
        if not any(o['uid'] in owned for o in p['metadata'].get('ownerReferences',[])):
            continue
        info={'name':p['metadata']['name'],'uid':p['metadata']['uid'],'owner':p['metadata'].get('ownerReferences'), 'deleting':p['metadata'].get('deletionTimestamp'),'phase':p['status']['phase'],'conditions':p['status'].get('conditions'),'containers':p['status'].get('containerStatuses'),'requested_images':[c['image'] for c in p['spec']['containers']]}
        if label != 'runtime-before':
            info['admin_health']={'result':'NOT_RUN','reason':'Baseline pods/proxy timeouts retained; localhost health collected separately, not collector-path acceptance'}
            result.append(info)
            continue
        if not info['deleting'] and info['phase']=='Running':
            info['admin_health']={}
            for endpoint in ('livez','readyz'):
                path='/api/v1/namespaces/labbit/pods/'+p['metadata']['name']+':9090/proxy/'+endpoint
                try:
                    body=kube('get','--raw',path,text=True,timeout=6)
                    info['admin_health'][endpoint]={'result':'HTTP_SUCCESS','body':body[:128]}
                except (RuntimeError,subprocess.TimeoutExpired):
                    info['admin_health'][endpoint]={'result':'NOT_VERIFIED'}
            try:
                metrics=kube('get','--raw','/api/v1/namespaces/labbit/pods/'+p['metadata']['name']+':9090/proxy/metrics',text=True,timeout=6)
                info['maintenance_metrics']=[line for line in metrics.splitlines() if line.startswith(('labbit_maintenance_','labbit_worker_control_'))]
                info['full_metrics_sha256']=hashlib.sha256(metrics.encode()).hexdigest()
            except (RuntimeError,subprocess.TimeoutExpired):
                info['metrics_result']='NOT_VERIFIED'
        result.append(info)
    apps={n:app(n) for n in APP_UIDS}
    cm=kube('get','configmap','backend-config','-n','labbit','-o','json')
    data={'at':now(),'deployment':{'uid':UID,'generation':d['metadata']['generation'],'status':d['status'],'spec_sha256':hashlib.sha256(json.dumps(d['spec'],sort_keys=True).encode()).hexdigest(),'image':d['spec']['template']['spec']['containers'][0]['image'],'env':d['spec']['template']['spec']['containers'][0].get('env'),'replicas':d['spec']['replicas']},'pods':result,'maintenance':cm['data'],'applications':{n:{'uid':v['metadata']['uid'],'spec':v['spec'],'health':v.get('status',{}).get('health'),'sync':v.get('status',{}).get('sync')} for n,v in apps.items()},'execution_gate':'HOLD','recovery_authorized':False,'cloud_verified':False}
    save(EVIDENCE,label+'.json',data)
    print(json.dumps({'label':label,'image':data['deployment']['image'],'status':d['status'],'pods':[{'name':p['name'],'phase':p['phase'],'deleting':p['deleting'],'health':p.get('admin_health')} for p in result]},ensure_ascii=False))


def local_health(label):
    runtime=load(EVIDENCE,'runtime-'+label+'.json')
    result=[]
    for p in runtime['pods']:
        if p['deleting'] or p['phase']!='Running':
            continue
        before=kube('get','pod',p['name'],'-n','labbit','-o','json')
        if before['metadata']['uid']!=p['uid'] or before['status']['containerStatuses']!=p['containers']:
            raise RuntimeError('POD_CHANGED_BEFORE_LOCAL_PROBE')
        probes={}
        for path in ('livez','readyz','metrics'):
            body=kube('exec','-n','labbit',p['name'],'-c','common-app','--','/bin/busybox','wget','-q','-T','3','-O','-','http://127.0.0.1:9090/'+path,text=True,timeout=10)
            probes[path]= {'body':body,'result':'HTTP_SUCCESS'} if path!='metrics' else {'maintenance_lines':[l for l in body.splitlines() if l.startswith(('labbit_maintenance_','labbit_worker_control_'))],'source_sha256':hashlib.sha256(body.encode()).hexdigest()}
        after=kube('get','pod',p['name'],'-n','labbit','-o','json')
        if before['metadata']['uid']!=after['metadata']['uid'] or before['status']['containerStatuses']!=after['status']['containerStatuses']:
            raise RuntimeError('POD_CHANGED_DURING_LOCAL_PROBE')
        result.append({'name':p['name'],'uid':p['uid'],'containers':p['containers'],'probes':probes})
    if len(result)!=2:
        raise RuntimeError('TWO_POD_LOCAL_PROBE_COVERAGE_MISSING')
    save(EVIDENCE,'local-health-'+label+'.json',{'at':now(),'method':'kubectl exec busybox wget to localhost:9090; no secret access','pods':result,'collector_pods_proxy_verified':False,'execution_gate':'HOLD','cloud_verified':False,'recovery_authorized':False})
    print(json.dumps({'label':label,'pods':[{'name':p['name'],'livez':p['probes']['livez']['body'],'readyz':p['probes']['readyz']['body'],'maintenance':p['probes']['metrics']['maintenance_lines']} for p in result]}))

    for n in APP_UIDS:
        if app(n)['spec']!=original[n]['spec']:
            raise RuntimeError('ORIGINAL_ARGO_SPEC_NOT_RESTORED')
    for kind,name,resource,path,creation in (
        ('Job',JOB,'job','/apis/batch/v1/namespaces/labbit/jobs/'+JOB,'migration-job-create'),
        ('SecretProviderClass',SPC,'secretproviderclass','/apis/secrets-store.csi.x-k8s.io/v1/namespaces/labbit/secretproviderclasses/'+SPC,'create-migration-spc')):
        x=kube('get',resource,name,'-n','labbit','-o','json')
        uid=load(EVIDENCE,creation+'.receipt.json')['result']['uid']
        if x['metadata']['uid']!=uid or x['metadata'].get('labels',{}).get('labbit.io/trial')!='pr196-20261123':
            raise RuntimeError('CLEANUP_OWNERSHIP_NOT_VERIFIED')
        if kind=='Job' and x.get('status',{}).get('succeeded')!=1:
            raise RuntimeError('MIGRATION_JOB_NOT_COMPLETED')
        options={'apiVersion':'v1','kind':'DeleteOptions','preconditions':{'uid':uid,'resourceVersion':x['metadata']['resourceVersion']},'propagationPolicy':'Foreground'}
        f=LOCAL/('delete-options-'+resource+'.json');f.write_text(json.dumps(options),encoding='utf-8')
        intent('cleanup-'+resource,{'name':name,'uid':uid,'delete_options':options})
        r=kube('delete','--raw',path,'-f',str(f))
        receipt('cleanup-'+resource,r)
    print('TRIAL_JOB_AND_SPC_DELETE_REQUESTED_WITH_UID_PRECONDITIONS')
