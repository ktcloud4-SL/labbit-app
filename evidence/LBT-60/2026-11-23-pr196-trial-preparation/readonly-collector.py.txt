def run(args):
    at=datetime.now(timezone.utc).isoformat()
    try:
        p=subprocess.run(args,capture_output=True,timeout=35)
        if p.returncode:
            # Only generic error category is public. No raw stdout/stderr saved.
            s=p.stderr.decode('utf8','replace')
            reason=next((x for x in ['AccessDenied','Forbidden','ImageNotFoundException','RepositoryNotFoundException','NotFound','Unauthorized','ExpiredToken','could not resolve','Unable to connect'] if x.lower() in s.lower()),'QUERY_NOT_VERIFIED')
            return dict(observed_at=at,exit_code=p.returncode,status='NOT_VERIFIED',reason=reason)
        raw=p.stdout
        try:v=json.loads(raw)
        except ValueError:v=raw.decode('utf8','replace').strip()
        return dict(observed_at=at,status='OBSERVED_READONLY',exit_code=0,raw_sha256=hashlib.sha256(raw).hexdigest(),value=v)
    except subprocess.TimeoutExpired:return dict(observed_at=at,status='NOT_VERIFIED',reason='TIMEOUT')

    if name=='applications':return [dict(name=x['metadata']['name'],uid=x['metadata']['uid'],source={k:x['spec'].get('source',{}).get(k) for k in ['repoURL','path','targetRevision']},syncPolicy=x['spec'].get('syncPolicy'),sync=x.get('status',{}).get('sync'),health=x.get('status',{}).get('health')) for x in v['items'] if 'backend' in x['metadata']['name'] or 'root' in x['metadata']['name']]
    if name=='cluster_identity':return dict(name=v['metadata']['name'],uid=v['metadata']['uid'])
    return v
